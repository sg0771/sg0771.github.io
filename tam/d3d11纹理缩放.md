#include <d3d11.h>
#include <d3dcompiler.h>
#include <wrl/client.h>

using Microsoft::WRL::ComPtr;

HRESULT ScaleRGB32Texture(
    ID3D11Device* device,
    ID3D11DeviceContext* context,
    ID3D11Texture2D* srcTexture,
    UINT dstWidth,
    UINT dstHeight,
    ID3D11Texture2D** outTexture)
{
    if (!device || !context || !srcTexture || !outTexture)
        return E_INVALIDARG;

    *outTexture = nullptr;

    D3D11_TEXTURE2D_DESC srcDesc{};
    srcTexture->GetDesc(&srcDesc);

    // RGB32 通常应是这两种之一
    if (srcDesc.Format != DXGI_FORMAT_R8G8B8A8_UNORM &&
        srcDesc.Format != DXGI_FORMAT_B8G8R8A8_UNORM)
    {
        return E_INVALIDARG;
    }

    ComPtr<ID3D11ShaderResourceView> srcSRV;
    HRESULT hr = device->CreateShaderResourceView(srcTexture, nullptr, &srcSRV);
    if (FAILED(hr))
        return hr;

    D3D11_TEXTURE2D_DESC dstDesc{};
    dstDesc.Width = dstWidth;
    dstDesc.Height = dstHeight;
    dstDesc.MipLevels = 1;
    dstDesc.ArraySize = 1;
    dstDesc.Format = srcDesc.Format;
    dstDesc.SampleDesc.Count = 1;
    dstDesc.Usage = D3D11_USAGE_DEFAULT;
    dstDesc.BindFlags = D3D11_BIND_RENDER_TARGET | D3D11_BIND_SHADER_RESOURCE;

    ComPtr<ID3D11Texture2D> dstTexture;
    hr = device->CreateTexture2D(&dstDesc, nullptr, &dstTexture);
    if (FAILED(hr))
        return hr;

    ComPtr<ID3D11RenderTargetView> dstRTV;
    hr = device->CreateRenderTargetView(dstTexture.Get(), nullptr, &dstRTV);
    if (FAILED(hr))
        return hr;

    static const char* shaderCode = R"(
        Texture2D srcTex : register(t0);
        SamplerState samp : register(s0);

        struct VSOut {
            float4 pos : SV_POSITION;
            float2 uv  : TEXCOORD0;
        };

        VSOut VSMain(uint id : SV_VertexID) {
            float2 pos[3] = {
                float2(-1.0, -1.0),
                float2(-1.0,  3.0),
                float2( 3.0, -1.0)
            };

            float2 uv[3] = {
                float2(0.0, 1.0),
                float2(0.0, -1.0),
                float2(2.0, 1.0)
            };

            VSOut o;
            o.pos = float4(pos[id], 0.0, 1.0);
            o.uv = uv[id];
            return o;
        }

        float4 PSMain(VSOut i) : SV_TARGET {
            return srcTex.Sample(samp, i.uv);
        }
    )";

    ComPtr<ID3DBlob> vsBlob;
    ComPtr<ID3DBlob> psBlob;
    ComPtr<ID3DBlob> errorBlob;

    hr = D3DCompile(shaderCode, strlen(shaderCode), nullptr, nullptr, nullptr,
        "VSMain", "vs_5_0", 0, 0, &vsBlob, &errorBlob);
    if (FAILED(hr))
        return hr;

    hr = D3DCompile(shaderCode, strlen(shaderCode), nullptr, nullptr, nullptr,
        "PSMain", "ps_5_0", 0, 0, &psBlob, &errorBlob);
    if (FAILED(hr))
        return hr;

    ComPtr<ID3D11VertexShader> vs;
    ComPtr<ID3D11PixelShader> ps;

    hr = device->CreateVertexShader(vsBlob->GetBufferPointer(), vsBlob->GetBufferSize(), nullptr, &vs);
    if (FAILED(hr))
        return hr;

    hr = device->CreatePixelShader(psBlob->GetBufferPointer(), psBlob->GetBufferSize(), nullptr, &ps);
    if (FAILED(hr))
        return hr;

    D3D11_SAMPLER_DESC samplerDesc{};
    samplerDesc.Filter = D3D11_FILTER_MIN_MAG_MIP_LINEAR;
    samplerDesc.AddressU = D3D11_TEXTURE_ADDRESS_CLAMP;
    samplerDesc.AddressV = D3D11_TEXTURE_ADDRESS_CLAMP;
    samplerDesc.AddressW = D3D11_TEXTURE_ADDRESS_CLAMP;
    samplerDesc.MaxLOD = D3D11_FLOAT32_MAX;

    ComPtr<ID3D11SamplerState> sampler;
    hr = device->CreateSamplerState(&samplerDesc, &sampler);
    if (FAILED(hr))
        return hr;

    D3D11_VIEWPORT vp{};
    vp.Width = static_cast<float>(dstWidth);
    vp.Height = static_cast<float>(dstHeight);
    vp.MinDepth = 0.0f;
    vp.MaxDepth = 1.0f;

    context->OMSetRenderTargets(1, dstRTV.GetAddressOf(), nullptr);
    context->RSSetViewports(1, &vp);

    context->IASetInputLayout(nullptr);
    context->IASetPrimitiveTopology(D3D11_PRIMITIVE_TOPOLOGY_TRIANGLELIST);

    context->VSSetShader(vs.Get(), nullptr, 0);
    context->PSSetShader(ps.Get(), nullptr, 0);
    context->PSSetShaderResources(0, 1, srcSRV.GetAddressOf());
    context->PSSetSamplers(0, 1, sampler.GetAddressOf());

    context->Draw(3, 0);

    ID3D11ShaderResourceView* nullSRV[] = { nullptr };
    ID3D11RenderTargetView* nullRTV[] = { nullptr };

    context->PSSetShaderResources(0, 1, nullSRV);
    context->OMSetRenderTargets(1, nullRTV, nullptr);

    *outTexture = dstTexture.Detach();
    return S_OK;
}
