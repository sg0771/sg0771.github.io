#include <iostream>
#include <vector>
#include <queue>
#include <cstring>
using namespace std;

const int MAXN = 100005;
const int INF = 0x3f3f3f3f;

vector<int> g[MAXN], rg[MAXN];
int n, m, q;
int mindis[MAXN], maxdis[MAXN];
int in[MAXN];
bool ring[MAXN], vis[MAXN];

// BFS求最短路
void bfs_min() {
    memset(mindis, 0x3f, sizeof(mindis));
    queue<int> q;
    mindis[1] = 1;
    q.push(1);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) {
            if (mindis[v] > mindis[u] + 1) {
                mindis[v] = mindis[u] + 1;
                q.push(v);
            }
        }
    }
}

// 拓扑排序判环 + 求最长路
void topo() {
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        in[i] = rg[i].size();
        if (in[i] == 0) q.push(i);
    }
    vector<int> topo_seq;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        topo_seq.push_back(u);
        for (int v : g[u]) {
            in[v]--;
            if (in[v] == 0) q.push(v);
        }
    }
    // 不在拓扑序列中的点属于环
    for (int i = 1; i <= n; ++i) {
        ring[i] = true;
    }
    for (int u : topo_seq) ring[u] = false;

    // 拓扑序求最长路
    memset(maxdis, 0, sizeof(maxdis));
    maxdis[1] = 1;
    for (int u : topo_seq) {
        if (maxdis[u] == 0) continue;
        for (int v : g[u]) {
            if (maxdis[v] < maxdis[u] + 1) {
                maxdis[v] = maxdis[u] + 1;
            }
        }
    }

    // 反向图传递环标记：能走到环的点也标记ring
    memset(vis, 0, sizeof(vis));
    queue<int> qr;
    for (int i = 1; i <= n; ++i) {
        if (ring[i]) {
            vis[i] = true;
            qr.push(i);
        }
    }
    while (!qr.empty()) {
        int u = qr.front(); qr.pop();
        for (int v : rg[u]) {
            if (!vis[v]) {
                vis[v] = true;
                ring[v] = true;
                qr.push(v);
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m >> q;
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        rg[v].push_back(u);
    }
    bfs_min();
    topo();
    while (q--) {
        int x, L;
        cin >> x >> L;
        if (mindis[x] == INF) {
            cout << "No\n";
            continue;
        }
        if (L < mindis[x]) {
            cout << "No\n";
            continue;
        }
        if (L == mindis[x]) {
            cout << "Yes\n";
            continue;
        }
        int delta = L - mindis[x];
        if (ring[x]) {
            if (delta % 2 == 0) cout << "Yes\n";
            else cout << "No\n";
        } else {
            if (L <= maxdis[x] && delta % 2 == 0) cout << "Yes\n";
            else cout << "No\n";
        }
    }
    return 0;
}
