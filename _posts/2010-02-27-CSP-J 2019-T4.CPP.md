# T4 加工零件(work)
## 题目描述

凯凯的工厂正在有条不紊地生产一种神奇的零件，神奇的零件的生产过程自然也很神奇。工厂里有n位工人，工人们从$1\sim n$编号。某些工人之间存在双向的零件传送带。保证每两名工人之间最多只存在一条传送带。

如果x号工人想生产一个被加工到第$L(L>1)$ 阶段的零件，则所有与 号工人有传送带直接相连的工人，都需要生产一个被加工到第$L-1$阶段的零件（但x号工人自己无需生产第$L-1$阶段的零件）。

如果x号工人想生产一个被加工到第1阶段的零件，则所有与x号工人有传送带直接相连的工人，都需要为x号工人提供一个原材料。

轩轩是1号工人。现在给出q张工单，第i张工单表示编号为 a<sub>i</sub> 的工人想生产一个第L<sub>i</sub>阶段的零件。

轩轩想知道对于每张工单，他是否需要给别人提供原材料。他知道聪明的你一定可以帮他计算出来！

### 输入格式
第一行三个正整数 m，n 和 q，分别表示工人的数目、传送带的数目和工单的数目。
接下来m 行，每行两个正整数u 和v ，表示编号为u 和v 的工人之间存在一条零件传输带。保证 $u!=v$。

接下来q 行，每行两个正整数a 和L ，表示编号为u 的工人想生产一个第L 阶段的零件。

### 输出格式
共q 行，每行一个字符串 Yes 或者 No 。如果按照第i 张工单生产，需要编号为 1 的轩轩提供原材料，则在第i 行输出 Yes ；否则在第i 行输出 No 。

~~~cpp
#include <iostream>
#include <vector>
#include <queue>
#include <cstring>
using namespace std;

const int MAXN = 100005;
const int INF = 0x3f3f3f3f;

vector<int> g[MAXN], rg[MAXN];
int n, m, q;
int mindis[MAXN], maxdis[MAXN];
int in[MAXN];
bool ring[MAXN], vis[MAXN];

// BFS求最短路
void bfs_min() {
    memset(mindis, 0x3f, sizeof(mindis));
    queue<int> q;
    mindis[1] = 1;
    q.push(1);
    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : g[u]) {
            if (mindis[v] > mindis[u] + 1) {
                mindis[v] = mindis[u] + 1;
                q.push(v);
            }
        }
    }
}

// 拓扑排序判环 + 求最长路
void topo() {
    queue<int> q;
    for (int i = 1; i <= n; ++i) {
        in[i] = rg[i].size();
        if (in[i] == 0) q.push(i);
    }
    vector<int> topo_seq;
    while (!q.empty()) {
        int u = q.front(); q.pop();
        topo_seq.push_back(u);
        for (int v : g[u]) {
            in[v]--;
            if (in[v] == 0) q.push(v);
        }
    }
    // 不在拓扑序列中的点属于环
    for (int i = 1; i <= n; ++i) {
        ring[i] = true;
    }
    for (int u : topo_seq) ring[u] = false;

    // 拓扑序求最长路
    memset(maxdis, 0, sizeof(maxdis));
    maxdis[1] = 1;
    for (int u : topo_seq) {
        if (maxdis[u] == 0) continue;
        for (int v : g[u]) {
            if (maxdis[v] < maxdis[u] + 1) {
                maxdis[v] = maxdis[u] + 1;
            }
        }
    }

    // 反向图传递环标记：能走到环的点也标记ring
    memset(vis, 0, sizeof(vis));
    queue<int> qr;
    for (int i = 1; i <= n; ++i) {
        if (ring[i]) {
            vis[i] = true;
            qr.push(i);
        }
    }
    while (!qr.empty()) {
        int u = qr.front(); qr.pop();
        for (int v : rg[u]) {
            if (!vis[v]) {
                vis[v] = true;
                ring[v] = true;
                qr.push(v);
            }
        }
    }
}

int main() {
    ios::sync_with_stdio(false);
    cin.tie(0);
    cin >> n >> m >> q;
    for (int i = 1; i <= m; ++i) {
        int u, v;
        cin >> u >> v;
        g[u].push_back(v);
        rg[v].push_back(u);
    }
    bfs_min();
    topo();
    while (q--) {
        int x, L;
        cin >> x >> L;
        if (mindis[x] == INF) {
            cout << "No\n";
            continue;
        }
        if (L < mindis[x]) {
            cout << "No\n";
            continue;
        }
        if (L == mindis[x]) {
            cout << "Yes\n";
            continue;
        }
        int delta = L - mindis[x];
        if (ring[x]) {
            if (delta % 2 == 0) cout << "Yes\n";
            else cout << "No\n";
        } else {
            if (L <= maxdis[x] && delta % 2 == 0) cout << "Yes\n";
            else cout << "No\n";
        }
    }
    return 0;
}
~~~
